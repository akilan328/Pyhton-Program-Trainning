2.Progressive Float Execution:
import struct
a=float(input())
a=struct.unpack('f',struct.pack('f',a))[0]
print(f"{a:.5f}")
print(f"{a:.4f}")
print(f"{a:.3f}")
print(f"{a:.2f}")
print(f"{a:.1f}")
print(f"{a:.0f}")
